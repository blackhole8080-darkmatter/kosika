It uses i386-elf-gcc and i386-elf-ld; use the commands in your rules instead.

int my_function() {
    return 0xbaba;
}

=== functioncalls.c ===
void caller() {
    my_func(0xdede);
}

int my_func(int arg) {
    return arg;
}

=== localvars.c ===
int my_function() {
    int my_var = 0xbaba;
    return my_var;
}

}
